# Sourcify v2 Integration Plan (M4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Use Sourcify v2 to detect proxy backdoors, decode any verified contract, surface contract-author notices, and flag suspiciously fresh deployments.

**Architecture:** Replace the legacy `/files/any/` Sourcify call with `/v2/contract/{chainId}/{address}?fields=...`. When the target is a proxy, recurse into its implementation and treat *that* as the load-bearing trust signal. When our hand-written recognizers miss but Sourcify has an ABI, fall back to a viem-based ABI decoder. Surface NatSpec `userdoc.notice` in the verdict checklist. Compute deployment age from Sourcify's `deployment.blockNumber` plus a chain-head RPC lookup.

**Tech Stack:** TypeScript, viem (`decodeFunctionData`, `getAbiItem`), `@intent-check/sourcify-client`, existing decoder/judge/extension packages, vitest mocks.

```ts
// packages/sourcify-client/src/v2.ts
export interface ProxyInfo {
  isProxy: boolean;
  proxyType?: string;
  implementationAddress?: string;
  resolutionError?: string;
}
export interface DeploymentInfo {
  blockNumber?: number;
  transactionHash?: string;
  deployer?: string;
}
export interface FunctionSignature { signature: string; selector4: string }
export interface UserDoc {
  notice?: string;
  methods?: Record<string, { notice?: string }>;
}
export interface VerifiedContractV2 {
  verified: boolean;
  matchType?: "exact_match" | "match";
  contractName?: string;
  abi: any[];
  proxy?: ProxyInfo;
  deployment?: DeploymentInfo;
  signatures?: { function: FunctionSignature[]; event: FunctionSignature[]; error: FunctionSignature[] };
  userdoc?: UserDoc;
  implementation?: VerifiedContractV2;
}

const FIELDS = ["abi", "compilation", "deployment", "proxyResolution", "signatures", "userdoc"].join(",");

export async function fetchVerifiedContractV2(args: {
  chainId: number; address: string; baseUrl?: string;
}): Promise<VerifiedContractV2> {
  const baseUrl = args.baseUrl ?? "https://sourcify.dev/server";
  const addr = args.address.toLowerCase();
  const url = `${baseUrl}/v2/contract/${args.chainId}/${addr}?fields=${FIELDS}`;
  const res = await fetch(url);
  if (res.status === 404) return { verified: false, abi: [] };
  if (!res.ok) return { verified: false, abi: [] };
  const body = (await res.json()) as any;

  const proxy = body.proxyResolution
    ? {
        isProxy: !!body.proxyResolution.isProxy,
        proxyType: body.proxyResolution.proxyType ?? undefined,
        implementationAddress: body.proxyResolution.implementations?.[0]?.address?.toLowerCase(),
        resolutionError: body.proxyResolution.proxyResolutionError?.message,
      }
    : undefined;

  const deployment = body.deployment
    ? {
        blockNumber: body.deployment.blockNumber ? Number(body.deployment.blockNumber) : undefined,
        transactionHash: body.deployment.transactionHash,
        deployer: body.deployment.deployer?.toLowerCase(),
      }
    : undefined;

  const sigs = body.signatures
    ? {
        function: (body.signatures.function ?? []).map((s: any) => ({ signature: s.signature, selector4: s.signatureHash4 })),
        event:    (body.signatures.event    ?? []).map((s: any) => ({ signature: s.signature, selector4: s.signatureHash4 })),
        error:    (body.signatures.error    ?? []).map((s: any) => ({ signature: s.signature, selector4: s.signatureHash4 })),
      }
    : undefined;

  return {
    verified: true,
    matchType: body.match,
    contractName: body.compilation?.contractName,
    abi: body.abi ?? [],
    proxy,
    deployment,
    signatures: sigs,
    userdoc: body.userdoc,
  };
}
```

```ts
// apps/extension/src/lib/blockTime.ts

/** Approximate blocks per day for our supported chains. Off by ~10% is fine —
 * we use this for "is this contract less than 7 days old?" thresholds, not
 * for displaying timestamps. */
const BLOCKS_PER_DAY: Record<number, number> = {
  1: 7200,        // Ethereum (12s/block)
  10: 17280,      // Optimism (5s/block)
  8453: 43200,    // Base (2s/block)
  42161: 345600,  // Arbitrum (250ms/block — but it's variable; this is an upper bound)
};

export function estimateAgeDays(args: { chainId: number; headBlock: number; deployBlock: number }): number | undefined {
  const bpd = BLOCKS_PER_DAY[args.chainId];
  if (!bpd) return undefined;
  return Math.max(0, (args.headBlock - args.deployBlock) / bpd);
}

/** Fetch chain head via JSON-RPC. Caller passes an RPC URL; we cache for 60s. */
const headCache = new Map<number, { head: number; at: number }>();
export async function getChainHead(chainId: number, rpcUrl: string): Promise<number | undefined> {
  const cached = headCache.get(chainId);
  if (cached && Date.now() - cached.at < 60_000) return cached.head;
  try {
    const res = await fetch(rpcUrl, {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ jsonrpc: "2.0", method: "eth_blockNumber", params: [], id: 1 }),
    });
    if (!res.ok) return undefined;
    const body = (await res.json()) as { result?: string };
    if (!body.result) return undefined;
    const head = Number(BigInt(body.result));
    headCache.set(chainId, { head, at: Date.now() });
    return head;
  } catch { return undefined; }
}

/** Free public RPCs we use only for `eth_blockNumber` reads — read-only,
 * no auth, no rate limit problem at our volume. Replace with a proxied
 * backend call before any production deploy. */

  1: "https://cloudflare-eth.com",
  10: "https://mainnet.optimism.io",
  8453: "https://mainnet.base.org",
  42161: "https://arb1.arbitrum.io/rpc",
};
```

```ts
// packages/decoder/src/recognizers/genericAbi.ts
import { decodeFunctionData } from "viem";
import type { DecodedAction } from "@intent-check/types";
import type { DecodeContext } from "../types"; // adjust import to match this package's local types

function stringifyArg(v: unknown): string {
  if (typeof v === "bigint") return v.toString();
  if (Array.isArray(v)) return `[${v.map(stringifyArg).join(", ")}]`;
  if (typeof v === "object" && v !== null) {
    return JSON.stringify(v, (_k, val) => (typeof val === "bigint" ? val.toString() : val));
  }
  return String(v);
}

export function tryDecodeWithAbi(input: DecodeContext, abi: any[]): DecodedAction | null {
  if (!abi || abi.length === 0) return null;
  if (!input.calldata || input.calldata.length < 10) return null;
  try {
    const decoded = decodeFunctionData({ abi, data: input.calldata as `0x${string}` });
    const fn = (abi as any[]).find(
      (item) => item.type === "function" && item.name === decoded.functionName
    );
    const argNames = fn?.inputs?.map((i: any) => i.name).filter(Boolean) ?? undefined;
    const args = (decoded.args ?? []).map(stringifyArg);
    return {
      kind: "generic",
      functionName: decoded.functionName,
      target: input.to,
      args,
      argNames,
      trusted: false,        // background flips this to true if registry hits
    };
  } catch {
    return null;
  }
}
```
